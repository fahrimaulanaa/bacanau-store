import { MetadataRoute } from 'next';

export default function robots(): MetadataRoute.Robots {
  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: ['/admin', '/checkout/pay'], // Blokir halaman rahasia
    },
    sitemap: 'https://bacanaustore.netlify.app/sitemap.xml',
  };
}
